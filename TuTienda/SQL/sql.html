USE TuTienda; GO /* TABLAS */ CREATE TABLE Roles ( Id INT IDENTITY(1,1) PRIMARY KEY, Nombre NVARCHAR(50) NOT NULL ); GO CREATE TABLE Usuarios ( Id INT IDENTITY(1,1) PRIMARY KEY, Nombres NVARCHAR(100) NOT NULL, Apellidos NVARCHAR(100) NOT NULL, Email NVARCHAR(150) NOT NULL, PasswordHash NVARCHAR(MAX) NOT NULL, RolId INT NOT NULL, Activo BIT NOT NULL DEFAULT 1, FechaRegistro DATETIME2 NOT NULL DEFAULT GETDATE(), NombreTienda NVARCHAR(150) NULL, DescripcionTienda NVARCHAR(500) NULL, CONSTRAINT FK_Usuarios_Roles FOREIGN KEY (RolId) REFERENCES Roles(Id) ); GO CREATE UNIQUE INDEX IX_Usuarios_Email ON Usuarios(Email); GO CREATE TABLE Categorias ( Id INT IDENTITY(1,1) PRIMARY KEY, Nombre NVARCHAR(100) NOT NULL, Descripcion NVARCHAR(300) NULL ); GO CREATE TABLE Productos ( Id INT IDENTITY(1,1) PRIMARY KEY, Nombre NVARCHAR(150) NOT NULL, Descripcion NVARCHAR(1000) NULL, Precio DECIMAL(10,2) NOT NULL, Stock INT NOT NULL, ImagenUrl NVARCHAR(500) NULL, CategoriaId INT NOT NULL, VendedorId INT NOT NULL, Activo BIT NOT NULL DEFAULT 1, FechaCreacion DATETIME2 NOT NULL DEFAULT GETDATE(), CONSTRAINT FK_Productos_Categorias FOREIGN KEY (CategoriaId) REFERENCES Categorias(Id), CONSTRAINT FK_Productos_Usuarios_Vendedor FOREIGN KEY (VendedorId) REFERENCES Usuarios(Id) ); GO CREATE TABLE Carritos ( Id INT IDENTITY(1,1) PRIMARY KEY, UsuarioId INT NULL, SessionId NVARCHAR(200) NULL, FechaCreacion DATETIME2 NOT NULL DEFAULT GETDATE(), CONSTRAINT FK_Carritos_Usuarios FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id) ); GO CREATE TABLE CarritoItems ( Id INT IDENTITY(1,1) PRIMARY KEY, CarritoId INT NOT NULL, ProductoId INT NOT NULL, Cantidad INT NOT NULL, PrecioUnitario DECIMAL(10,2) NOT NULL, CONSTRAINT FK_CarritoItems_Carritos FOREIGN KEY (CarritoId) REFERENCES Carritos(Id) ON DELETE CASCADE, CONSTRAINT FK_CarritoItems_Productos FOREIGN KEY (ProductoId) REFERENCES Productos(Id) ); GO CREATE TABLE Pedidos ( Id INT IDENTITY(1,1) PRIMARY KEY, GrupoCompraId UNIQUEIDENTIFIER NOT NULL, ClienteId INT NOT NULL, VendedorId INT NOT NULL, FechaPedido DATETIME2 NOT NULL DEFAULT GETDATE(), Estado INT NOT NULL DEFAULT 0, -- 0=Pendiente,1=Confirmado,2=Entregado,3=Cancelado Total DECIMAL(10,2) NOT NULL, MetodoPagoSimulado NVARCHAR(100) NOT NULL DEFAULT 'Pago simulado', CONSTRAINT FK_Pedidos_Usuarios_Cliente FOREIGN KEY (ClienteId) REFERENCES Usuarios(Id), CONSTRAINT FK_Pedidos_Usuarios_Vendedor FOREIGN KEY (VendedorId) REFERENCES Usuarios(Id) ); GO CREATE TABLE DetallesPedido ( Id INT IDENTITY(1,1) PRIMARY KEY, PedidoId INT NOT NULL, ProductoId INT NOT NULL, Cantidad INT NOT NULL, PrecioUnitario DECIMAL(10,2) NOT NULL, Subtotal DECIMAL(10,2) NOT NULL, CONSTRAINT FK_DetallesPedido_Pedidos FOREIGN KEY (PedidoId) REFERENCES Pedidos(Id) ON DELETE CASCADE, CONSTRAINT FK_DetallesPedido_Productos FOREIGN KEY (ProductoId) REFERENCES Productos(Id) ); GO /* STORED PROCEDURES*/ -- Filtro + paginación de productos (catálogo y búsqueda) CREATE OR ALTER PROCEDURE dbo.sp_FiltrarProductosPaginados @nombre NVARCHAR(150) = '', @precioMin DECIMAL(10,2) = 0, @categoriaId INT = 0, -- 0 = todas las categorías @paginaActual INT = 1, @elementosPorPagina INT = 8 AS BEGIN SET NOCOUNT ON; DECLARE @inicio INT = (@paginaActual - 1) * @elementosPorPagina; SELECT p.Id, p.Nombre, p.Descripcion, p.Precio, p.Stock, p.ImagenUrl, p.CategoriaId, p.VendedorId, p.Activo, COUNT(*) OVER() AS TotalRegistros FROM Productos p WHERE (p.Nombre LIKE '%' + @nombre + '%' OR @nombre = '') AND p.Precio >= @precioMin AND (p.CategoriaId = @categoriaId OR @categoriaId = 0) AND p.Activo = 1 ORDER BY p.Id OFFSET @inicio ROWS FETCH NEXT @elementosPorPagina ROWS ONLY; END GO -- Reporte 1: Ventas por período (agrupado por día) CREATE OR ALTER PROCEDURE dbo.sp_ReporteVentasPorPeriodo @fechaInicio DATE, @fechaFin DATE AS BEGIN SET NOCOUNT ON; SELECT CAST(p.FechaPedido AS DATE) AS Fecha, COUNT(DISTINCT p.Id) AS CantidadPedidos, SUM(p.Total) AS Total FROM Pedidos p WHERE CAST(p.FechaPedido AS DATE) BETWEEN @fechaInicio AND @fechaFin AND p.Estado <> 3 GROUP BY CAST(p.FechaPedido AS DATE) ORDER BY Fecha; END GO -- Reporte 2: Productos más vendidos CREATE OR ALTER PROCEDURE dbo.sp_ReporteProductosMasVendidos @fechaInicio DATE = NULL, @fechaFin DATE = NULL, @topN INT = 10 AS BEGIN SET NOCOUNT ON; SELECT TOP (@topN) pr.Id AS ProductoId, pr.Nombre, SUM(dp.Cantidad) AS CantidadVendida, SUM(dp.Subtotal) AS Ingresos FROM DetallesPedido dp INNER JOIN Pedidos pe ON pe.Id = dp.PedidoId INNER JOIN Productos pr ON pr.Id = dp.ProductoId WHERE pe.Estado <> 3 AND (@fechaInicio IS NULL OR CAST(pe.FechaPedido AS DATE) >= @fechaInicio) AND (@fechaFin IS NULL OR CAST(pe.FechaPedido AS DATE) <= @fechaFin) GROUP BY pr.Id, pr.Nombre ORDER BY CantidadVendida DESC; END GO -- Reporte 3: Ingresos por vendedor CREATE OR ALTER PROCEDURE dbo.sp_ReporteIngresosPorVendedor @fechaInicio DATE = NULL, @fechaFin DATE = NULL AS BEGIN SET NOCOUNT ON; SELECT u.Id AS VendedorId, COALESCE(u.NombreTienda, u.Nombres + ' ' + u.Apellidos) AS NombreVendedor, COUNT(DISTINCT p.Id) AS CantidadPedidos, SUM(p.Total) AS Total FROM Pedidos p INNER JOIN Usuarios u ON u.Id = p.VendedorId WHERE p.Estado <> 3 AND (@fechaInicio IS NULL OR CAST(p.FechaPedido AS DATE) >= @fechaInicio) AND (@fechaFin IS NULL OR CAST(p.FechaPedido AS DATE) <= @fechaFin) GROUP BY u.Id, u.NombreTienda, u.Nombres, u.Apellidos ORDER BY Total DESC; END GO -- Reporte 4: Comparación de ventas mes a mes CREATE OR ALTER PROCEDURE dbo.sp_ReporteComparacionMensual @anio INT AS BEGIN SET NOCOUNT ON; SELECT MONTH(p.FechaPedido) AS Mes, COUNT(DISTINCT p.Id) AS CantidadPedidos, SUM(p.Total) AS Total FROM Pedidos p WHERE YEAR(p.FechaPedido) = @anio AND p.Estado <> 3 GROUP BY MONTH(p.FechaPedido) ORDER BY Mes;
